# OS-fliken steg 1 — grunden (databas, os-ingest, insamlare) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ett lokalt Node-skript samlar appar, skills, MCP:er och plugins och skickar en ögonblicksbild till Supabase, där bara Mats kan läsa den.

**Architecture:** `tools/os-collector/` (Node ≥18, inga beroenden) läser registret, MCP Inventory, `~/.claude`-konfig och repona → bygger poster via vitlistor → POST till Edge Function `os-ingest` med header `x-os-secret` → upsert i `os_items` + radering av saknade per typ → kvitto i `os_collect_log`. Samma mönster som `halsa-ingest`.

**Tech Stack:** Node 18+ (`node:test`, `node:fs`, `node:child_process`), Deno Edge Function, Postgres/Supabase Vault.

## Global Constraints

- Supabase-projekt: "Mats Project" (`ncgxerxkgoxptcwvramn`). Namespace `os_`. Inga ändringar i andra tabeller/funktioner.
- RLS läs: `(auth.jwt()->>'email') = 'mats@enmarks.se'`. Inga skrivpolicies.
- `deploy_edge_function`: alltid `verify_jwt: false`. Auth sker i koden.
- Ingen minnescache i edge-funktionen (modulvariabler delas inte mellan anrop).
- Insamlaren skickar aldrig: MCP `env`/`headers`/`args`, filinnehåll, källkod, nycklar. Vitlista per typ.
- Lokal hemlighet: `~/.os-ingest/secret` (utanför repot, printas aldrig).
- Repot är publikt: inga personliga värden (nycklar, e-post utöver ägar-konstanten som redan finns i koden, interna id:n) i committad kod eller fixtures.
- SQL körs via `~/.supabase/mats_project.conninfo` + psycopg2 (eller Supabase-MCP `apply_migration`). Skriv aldrig ut conninfo.

```ts
// os-ingest — tar emot insamlarens ögonblicksbild av AI Native OS.
// Kontrakt: docs/superpowers/specs/2026-10-12-os-flik-design.md
// Auth: header "x-os-secret" mot två Vault-nycklar (current + previous), konstant tid.
// Semantik: varje kind i body.kinds ersätts helt — upsert av skickade poster, radering av övriga i samma kind.
import { createClient } from "npm:@supabase/supabase-js@2";

const supabase = createClient(Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!);
const KINDS = new Set(["app", "skill", "mcp", "plugin", "run"]);
const MAX_BODY_BYTES = 5 * 1024 * 1024;

function constantTimeEq(a: string, b: string): boolean {
  const enc = new TextEncoder(); const ab = enc.encode(a), bb = enc.encode(b);
  if (ab.length !== bb.length) return false;
  let diff = 0; for (let i = 0; i < ab.length; i++) diff |= ab[i] ^ bb[i];
  return diff === 0;
}
const json = (status: number, body: unknown) =>
  new Response(JSON.stringify(body), { status, headers: { "Content-Type": "application/json" } });

Deno.serve(async (req) => {
  if (req.method !== "POST") return json(405, { error: "method not allowed" });
  const provided = req.headers.get("x-os-secret") ?? "";
  if (!provided) return json(401, { error: "unauthorized" });
  const { data: secrets, error: sErr } = await supabase.rpc("os_get_ingest_secrets");
  if (sErr || !secrets?.length) return json(503, { error: "auth backend unavailable" });
  if (!secrets.some((r: { secret: string }) => r.secret && constantTimeEq(provided, r.secret)))
    return json(401, { error: "unauthorized" });

  const text = await req.text();
  if (text.length > MAX_BODY_BYTES) return json(413, { error: "payload too large" });
  let body: { kinds?: unknown; items?: unknown };
  try { body = JSON.parse(text); } catch { return json(400, { error: "invalid json" }); }
  const kinds = Array.isArray(body.kinds) ? body.kinds.filter((k) => KINDS.has(k as string)) as string[] : [];
  const items = Array.isArray(body.items) ? body.items : [];
  if (!kinds.length) return json(400, { error: "no valid kinds" });
  const rows = [];
  for (const it of items as Record<string, unknown>[]) {
    if (!it || !kinds.includes(it.kind as string) || typeof it.id !== "string" || !it.id
        || typeof it.data !== "object" || it.data === null) return json(400, { error: "invalid item" });
    rows.push({ kind: it.kind, id: it.id, data: it.data, collected_at: new Date().toISOString() });
  }

  const { data: log } = await supabase.from("os_collect_log").insert({ status: "running" }).select("id").single();
  const errs: string[] = [];
  let removed = 0;
  if (rows.length) {
    const { error } = await supabase.from("os_items").upsert(rows, { onConflict: "kind,id" });
    if (error) errs.push(error.message);
  }
  if (!errs.length) {
    for (const k of kinds) {
      const keep = rows.filter((r) => r.kind === k).map((r) => r.id as string);
      let q = supabase.from("os_items").delete({ count: "exact" }).eq("kind", k);
      if (keep.length) q = q.not("id", "in", `(${keep.map((id) => `"${id.replace(/"/g, '\\"')}"`).join(",")})`);
      const { error, count } = await q;
      if (error) errs.push(error.message); else removed += count ?? 0;
    }
  }
  const counts = Object.fromEntries(kinds.map((k) => [k, rows.filter((r) => r.kind === k).length]));
  const status = errs.length ? "error" : "ok";
  if (log?.id) await supabase.from("os_collect_log").update({
    completed_at: new Date().toISOString(), status, counts,
    error_summary: errs.join(" | ").slice(0, 500) || null,
  }).eq("id", log.id);
  return json(errs.length ? 500 : 200, { status, upserted: rows.length, removed });
});
```

```js
// Markdown-tabeller ur AI Native OS-registren. Hanterar \| inne i [[länk\|alias]].
const splitRow = (line) =>
  line.trim().replace(/^\||\|$/g, '').split(/(?<!\\)\|/).map((c) => c.trim().replace(/\\\|/g, '|'));
const isSep = (line) => /^\s*\|?\s*:?-{3,}/.test(line);

export function parseTables(md) {
  const lines = md.split(/\r?\n/), out = [];
  for (let i = 0; i < lines.length - 1; i++) {
    if (!lines[i].trim().startsWith('|') || !isSep(lines[i + 1])) continue;
    const headers = splitRow(lines[i]), rows = [];
    let j = i + 2;
    for (; j < lines.length && lines[j].trim().startsWith('|'); j++) {
      const cells = splitRow(lines[j]);
      rows.push(Object.fromEntries(headers.map((h, k) => [h, cells[k] ?? ''])));
    }
    out.push({ headers, rows });
    i = j - 1;
  }
  return out;
}

export const clean = (s) => (s ?? '')
  .replace(/\[\[[^\]|]*\|([^\]]+)\]\]/g, '$1')
  .replace(/\[\[([^\]]+)\]\]/g, (_, p) => p.split('/').pop())
  .replace(/\*\*|`/g, '')
  .replace(/\s+/g, ' ').trim();
```

```js
import { readFileSync, readdirSync, existsSync, statSync } from 'node:fs';
import { join } from 'node:path';
import { parseTables, clean } from './md.mjs';

const read = (p) => readFileSync(p, 'utf8').replace(/^\uFEFF/, '');
const frontmatter = (text) => {
  const m = text.match(/^---\r?\n([\s\S]*?)\r?\n---/);
  if (!m) return {};
  const out = {};
  for (const line of m[1].split(/\r?\n/)) {
    const kv = line.match(/^(\w[\w-]*):\s*(.*)$/);
    if (kv) out[kv[1]] = kv[2].replace(/^["']|["']$/g, '');
  }
  return out;
};
const firstSentence = (s) => (s.match(/^.*?[.!?](\s|$)/)?.[0] ?? s).trim().slice(0, 200);
const dirs = (p) => existsSync(p) ? readdirSync(p).filter((d) => statSync(join(p, d)).isDirectory()) : [];

export function skills(sources) {
  const out = [];
  for (const { path, origin } of sources) {
    for (const d of dirs(path)) {
      const f = join(path, d, 'SKILL.md');
      if (!existsSync(f)) continue;
      const fm = frontmatter(read(f));
      const name = fm.name || d;
      out.push({ kind: 'skill', id: `${origin}:${name}`, data: { name, summary: firstSentence(fm.description ?? ''), origin } });
    }
  }
  return out;
}

export function plugins(installedPath) {
  const { plugins: all = {} } = JSON.parse(read(installedPath));
  return Object.entries(all).map(([key, [inst]]) => {
    const [name, marketplace] = key.split('@');
    const p = inst.installPath;
    return { kind: 'plugin', id: key, data: {
      name, marketplace, version: inst.version, updated: inst.lastUpdated,
      skills: dirs(join(p, 'skills')).length,
      agents: existsSync(join(p, 'agents')) ? readdirSync(join(p, 'agents')).filter((f) => f.endsWith('.md')).length : 0,
      hooks: existsSync(join(p, 'hooks')),
    } };
  });
}

const slug = (s) => clean(s).toLowerCase().replace(/\s*\(.*\)\s*/g, '').trim();

export function mcps(claudeJsonPath, inventoryPath) {
  const byId = new Map();
  const { mcpServers = {} } = JSON.parse(read(claudeJsonPath));
  for (const [name, cfg] of Object.entries(mcpServers)) {
    const data = { name, transport: cfg.url ? 'http' : 'stdio' };
    if (cfg.url) data.host = new URL(cfg.url).hostname;   // vitlista: bara värdnamn
    byId.set(name.toLowerCase(), { kind: 'mcp', id: name.toLowerCase(), data });
  }
  for (const t of parseTables(read(inventoryPath)).filter((t) => t.headers[0] === 'MCP')) {
    for (const r of t.rows) {
      const id = slug(r.MCP);
      const prev = byId.get(id)?.data ?? { name: clean(r.MCP), transport: 'connector' };
      byId.set(id, { kind: 'mcp', id, data: { ...prev,
        status: clean(r.Status), purpose: clean(r['Primärt användningsområde']),
        consumers: clean(r.Konsumenter), risk: clean(r.Risk), verified: clean(r.Verifierad).slice(0, 10) } });
    }
  }
  return [...byId.values()];
}
```

```js
import { readFileSync, readdirSync, existsSync, statSync } from 'node:fs';
import { join, extname } from 'node:path';
import { execFileSync } from 'node:child_process';

const KEY_DEPS = ['next', 'react', 'vite', 'tailwindcss', 'firebase', '@supabase/supabase-js', '@neondatabase/serverless', 'expo'];
const SKIP = new Set(['node_modules', '.git', '.next', 'build', 'dist', '.dart_tool', 'quality-runs', '.vercel']);
const CODE_EXT = new Set(['.js', '.mjs', '.ts', '.tsx', '.jsx', '.html', '.dart', '.kt']);

export function stackOf(dir) {
  const out = [];
  const pj = join(dir, 'package.json');
  if (existsSync(pj)) {
    const p = JSON.parse(readFileSync(pj, 'utf8'));
    const deps = { ...p.dependencies, ...p.devDependencies };
    for (const d of KEY_DEPS) if (deps[d]) out.push(`${d} ${deps[d].replace(/^[\^~]/, '')}`);
  }
  if (existsSync(join(dir, 'pubspec.yaml'))) out.push('flutter');
  if (existsSync(join(dir, 'build.gradle.kts')) || existsSync(join(dir, 'app', 'build.gradle.kts'))) out.push('kotlin');
  if (!out.length && existsSync(join(dir, 'index.html'))) out.push('vanilla');
  return out;
}

function* files(dir, depth = 0) {
  if (depth > 5 || !existsSync(dir)) return;
  for (const f of readdirSync(dir)) {
    if (SKIP.has(f)) continue;
    const p = join(dir, f), st = statSync(p);
    if (st.isDirectory()) yield* files(p, depth + 1);
    else if (CODE_EXT.has(extname(f)) && st.size < 2_000_000) yield p;
  }
}

export function dbsInCode(dir) {
  const found = new Set();
  for (const f of files(dir)) {
    const t = readFileSync(f, 'utf8');
    for (const m of t.matchAll(/https:\/\/([a-z0-9]{20})\.supabase\.co/g)) found.add(`supabase:${m[1]}`);
    for (const m of t.matchAll(/projectId:\s*["']([a-z0-9-]+)["']/g)) found.add(`firebase:${m[1]}`);
    if (/neon\.tech|@neondatabase\/serverless/.test(t)) found.add('neon');
  }
  return [...found].sort();
}

export function gitInfo(dir) {
  try {
    const g = (...a) => execFileSync('git', ['-C', dir, ...a], { encoding: 'utf8', stdio: ['ignore', 'pipe', 'ignore'] }).trim();
    const [commit, date] = g('log', '-1', '--format=%h|%cI').split('|');
    return { commit, date, branch: g('rev-parse', '--abbrev-ref', 'HEAD'), dirty: g('status', '--porcelain').length > 0 };
  } catch { return null; }
}

export const liveUrl = (cell) => cell.match(/https:\/\/[^\s`|)]+/)?.[0] ?? null;
```

```js
#!/usr/bin/env node
// OS-insamlaren — se README.md. Skickar aldrig hemligheter; varje källa har vitlista.
import { readFileSync, existsSync, readdirSync } from 'node:fs';
import { join, dirname, resolve } from 'node:path';
import { homedir } from 'node:os';
import { fileURLToPath } from 'node:url';
import { apps, skills, plugins, mcps } from './sources.mjs';

const HERE = dirname(fileURLToPath(import.meta.url));
const ROOT = resolve(HERE, '../../../..');               // Claude Co-Work OS
const OS = join(ROOT, 'AI Native OS');
const CLAUDE = join(homedir(), '.claude');
const ENDPOINT = 'https://ncgxerxkgoxptcwvramn.supabase.co/functions/v1/os-ingest';

async function ping(url) {
  try {
    const r = await fetch(url, { method: 'HEAD', redirect: 'follow', signal: AbortSignal.timeout(8000) });
    return { ok: r.status < 400, code: r.status, at: new Date().toISOString() };
  } catch { return { ok: false, code: 0, at: new Date().toISOString() }; }
}

const installed = JSON.parse(readFileSync(join(CLAUDE, 'plugins', 'installed_plugins.json'), 'utf8')).plugins ?? {};
const skillSources = [
  { path: join(CLAUDE, 'skills'), origin: 'egen' },
  ...Object.entries(installed).map(([k, [i]]) => ({ path: join(i.installPath, 'skills'), origin: `plugin:${k.split('@')[0]}` })),
];

const appItems = apps(join(OS, '10_REGISTRY', 'Projects Registry.md'), ROOT);
await Promise.all(appItems.map(async (a) => { if (a.data.liveUrl) a.data.live = await ping(a.data.liveUrl); }));

const items = [
  ...appItems,
  ...skills(skillSources),
  ...plugins(join(CLAUDE, 'plugins', 'installed_plugins.json')),
  ...mcps(join(homedir(), '.claude.json'), join(OS, '07_MCP', 'MCP Inventory.md')),
];
const body = { kinds: ['app', 'skill', 'mcp', 'plugin'], items };

if (process.argv.includes('--dry')) { console.log(JSON.stringify(body, null, 1)); process.exit(0); }

const secret = readFileSync(join(homedir(), '.os-ingest', 'secret'), 'utf8').trim();
const res = await fetch(ENDPOINT, { method: 'POST', headers: { 'Content-Type': 'application/json', 'x-os-secret': secret }, body: JSON.stringify(body) });
const out = await res.text();
console.log(res.status, out);
process.exit(res.ok ? 0 : 1);
```
